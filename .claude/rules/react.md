# React

## Blank lines

Leave a blank line between imports, the component declaration, state, handlers and the JSX:

```tsx
import { useState } from "react";
↕
const Counter = () => {
  const [count, setCount] = useState<number>(0);
  const [other, setOther] = useState<number>(0);
  ↕
  const handleClick = () => {
    setCount((count) => count + 1);
  };
  ↕
  return <div>{count}</div>;
};
↕
export default Counter;
```

## Export default

Write components as an **arrow function** with **export default** at the bottom of the file:

```tsx
const Component = () => {
  // ...
};

export default Component; ✅✅✅
```

To export other things alongside it:

```tsx
const Component = () => {
  // ...
};

export { Component as default, OtherComponent };
```

Or:

```tsx
const Component = () => {
  // ...
};

export default Component;
export { OtherComponent };
```

> Exception: components in `components/ui` are generated by shadcn/ui and use **named exports** (`export { Button, buttonVariants }`). Leave them as they are.

## Naming event handlers

Prefix handlers with **handle**, followed by the event name:

```ts
const handleClick = () => {};

const handleChange = () => {};
```

A handler passed through _props_ is prefixed with **on** instead:

```tsx
import { Button } from "@/components/ui/button";

interface Props {
  onClose: () => void;
}

const EnhancedButton = (props: Props) => {
  const { onClose } = props;

  return <Button onClick={onClose}>Click me!</Button>;
};

export default EnhancedButton;
```

Used as:

```tsx
const Parent = () => {
  const handleClose = () => {
    // ...
  };

  return <EnhancedButton onClose={handleClose} />;
};
```

## Props

Do not destructure props in the parameter list. Take `props` and destructure at the _top level_ of the component:

```tsx
import { Button } from "@/components/ui/button";

interface Props {
  title: string;
  onClose: () => void;
}

const Component = (props: Props) => {
  const { title, onClose } = props; // ✅✅✅

  return <Button onClick={onClose}>{title}</Button>;
};

export default Component;
```

When props include _children_, type the component with **FCC** from `@/types/react`:

```tsx
import { Button } from "@/components/ui/button";
import type { FCC } from "@/types/react";

interface Props {
  onClose: () => void;
}

const Component: FCC<Props> = (props) => {
  const { children, onClose } = props;

  return <Button onClick={onClose}>{children}</Button>;
};

export default Component;
```

For stricter cases — a layout component that requires exactly two children, say — declare _children_ as a _tuple_:

```tsx
import type { ReactNode } from "react";
import type { FCC } from "@/types/react";

interface Props {
  children: [ReactNode, ReactNode];
}

const Component: FCC<Props> = (props) => {
  const { children } = props;

  return <div className="grid h-full grid-rows-[auto_1fr]">{children}</div>;
};

export default Component;
```
